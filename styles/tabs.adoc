/* Tabbed Component Styles */

/* Container */
.tabs-container {
  margin: 1.5rem 0;
}

/* Hide radio buttons */
.tabs-container input[type="radio"] {
  display: none;
}

/* Tab header container */
.tabs-header {
  display: flex;
  border-bottom: 1px solid #d0d0d0;
}

/* Tab labels (buttons) */
.tabs-container label {
  padding: 0.75rem 1.5rem;
  cursor: pointer;
  user-select: none;

  border: 1px solid #d0d0d0;
  border-bottom: none;
  margin-right: -1px;

  opacity: 0.7;
  transition: opacity 0.2s ease, box-shadow 0.2s ease;
}

.tabs-container label:hover {
  opacity: 1;
  box-shadow: inset 0 2px 0 0 rgba(0, 0, 0, 0.1);
}

/* Active tab styling
   Matches checked radio buttons with labels */
.tabs-container input[type="radio"]:nth-of-type(1):checked ~ .tabs-header label:nth-of-type(1),
.tabs-container input[type="radio"]:nth-of-type(2):checked ~ .tabs-header label:nth-of-type(2),
.tabs-container input[type="radio"]:nth-of-type(3):checked ~ .tabs-header label:nth-of-type(3),
.tabs-container input[type="radio"]:nth-of-type(4):checked ~ .tabs-header label:nth-of-type(4),
.tabs-container input[type="radio"]:nth-of-type(5):checked ~ .tabs-header label:nth-of-type(5),
.tabs-container input[type="radio"]:nth-of-type(6):checked ~ .tabs-header label:nth-of-type(6),
.tabs-container input[type="radio"]:nth-of-type(7):checked ~ .tabs-header label:nth-of-type(7),
.tabs-container input[type="radio"]:nth-of-type(8):checked ~ .tabs-header label:nth-of-type(8),
.tabs-container input[type="radio"]:nth-of-type(9):checked ~ .tabs-header label:nth-of-type(9),
.tabs-container input[type="radio"]:nth-of-type(10):checked ~ .tabs-header label:nth-of-type(10) {
  opacity: 1;
  font-weight: 600;

  position: relative;
  z-index: 1;

  box-shadow: inset 0 3px 0 0 currentColor;
}

/* Tab content container */
.tabs-content {
  border: 1px solid #d0d0d0;
  border-top: none;
}

/* Individual tab panels */
.tab-panel {
  display: none;
  padding: 1.5rem;
}

/* Show active tab panel
   Matches checked radio buttons with panels */
.tabs-container input[type="radio"]:nth-of-type(1):checked ~ .tabs-content .tab-panel:nth-of-type(1),
.tabs-container input[type="radio"]:nth-of-type(2):checked ~ .tabs-content .tab-panel:nth-of-type(2),
.tabs-container input[type="radio"]:nth-of-type(3):checked ~ .tabs-content .tab-panel:nth-of-type(3),
.tabs-container input[type="radio"]:nth-of-type(4):checked ~ .tabs-content .tab-panel:nth-of-type(4),
.tabs-container input[type="radio"]:nth-of-type(5):checked ~ .tabs-content .tab-panel:nth-of-type(5),
.tabs-container input[type="radio"]:nth-of-type(6):checked ~ .tabs-content .tab-panel:nth-of-type(6),
.tabs-container input[type="radio"]:nth-of-type(7):checked ~ .tabs-content .tab-panel:nth-of-type(7),
.tabs-container input[type="radio"]:nth-of-type(8):checked ~ .tabs-content .tab-panel:nth-of-type(8),
.tabs-container input[type="radio"]:nth-of-type(9):checked ~ .tabs-content .tab-panel:nth-of-type(9),
.tabs-container input[type="radio"]:nth-of-type(10):checked ~ .tabs-content .tab-panel:nth-of-type(10) {
  display: block;
}

/* Responsive adjustments */
@media (max-width: 768px) {
  .tabs-container label {
    padding: 0.5rem 1rem;
  }

  .tab-panel {
    padding: 1rem;
  }
}
